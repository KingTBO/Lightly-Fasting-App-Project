-- Sketch only. Runs in the same Supabase project; nothing is applied until you approve.
create table groups (
  id uuid primary key default gen_random_uuid(),
  name text not null check (char_length(name) between 1 and 32),
  kind text not null default 'friends' check (kind in ('friends','family','ramadan')),
  owner_id uuid not null references auth.users on delete cascade,
  max_members int not null default 10
);
create table group_members (
  group_id uuid references groups on delete cascade,
  user_id  uuid references auth.users on delete cascade,
  role text not null default 'member' check (role in ('owner','member')),
  sharing boolean not null default true,
  primary key (group_id, user_id)
);
create table my_status (
  user_id uuid primary key references auth.users on delete cascade,
  state text not null check (state in ('fasting','resting','idle')),
  started_at timestamptz, goal_hours numeric check (goal_hours between 1 and 48),
  updated_at timestamptz not null default now()
);  -- RLS: select/insert/update only where user_id = auth.uid()

create function is_group_member(gid uuid) returns boolean
language sql stable security definer set search_path = '' as $$
  select exists (select 1 from public.group_members where group_id = gid and user_id = auth.uid());
$$;

create function group_board(gid uuid, d date) returns table (
  user_id uuid, name text, avatar text, status text, progress real, cheered boolean
) language sql stable security definer set search_path = '' as $$
  -- returns 'fasting' | 'done' | 'part' | 'rest' | 'none' only. If a member's sharing = false: 'rest'.
  -- progress for today = least(1, hours since started_at / goal_hours), computed here.
  select ... where public.is_group_member(gid);
$$;
